Sample Queries & Frameworks

SQL CTE Query

Inventory Shrinkage Root-Cause CTE

Shrinkage Reduced
-57%
WITH InventoryVariance AS (
  SELECT 
    p.sku_code,
    p.category_name,
    SUM(p.system_stock_units) AS book_stock,
    SUM(c.physical_count_units) AS physical_stock,
    (SUM(p.system_stock_units) - SUM(c.physical_count_units)) AS variance_units,
    ROUND(((SUM(p.system_stock_units) - SUM(c.physical_count_units))::numeric 
      / NULLIF(SUM(p.system_stock_units), 0)) * 100, 2) AS shrinkage_pct
  FROM retail_products p
  JOIN inventory_cycle_counts c ON p.product_id = c.product_id
  WHERE c.audit_date >= CURRENT_DATE - INTERVAL '6 months'
  GROUP BY p.sku_code, p.category_name
)
SELECT sku_code, category_name, book_stock, physical_stock, variance_units, shrinkage_pct
FROM InventoryVariance
WHERE shrinkage_pct > 2.0
ORDER BY variance_units DESC;
